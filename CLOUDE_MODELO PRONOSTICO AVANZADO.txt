# ─────────────────────────────────────────────────────────────────────────────
# CELDA 4 — PREPARACIÓN DE LA SERIE TEMPORAL
# Filtramos los días con cero producción (cierre operativo, no son outliers).
# Creamos la variable objetivo: DEMANDA = VENTAS + DEVOLUCIONES
# La demanda real es lo que el mercado quería comprar ese día, independientemente
# de cuánto se produjo.
# ─────────────────────────────────────────────────────────────────────────────

# Convertir FECHA a datetime
df_raw['FECHA'] = pd.to_datetime(df_raw['FECHA'])

# Ordenar por fecha
df_raw = df_raw.sort_values('FECHA').reset_index(drop=True)

# Identificar y separar días de cierre operativo (producción = 0)
# Nota: El 29/02/2024 es un cero aislado → posible dato faltante
# Los días 23-30/09/2025 son cierre confirmado por el usuario
dias_cierre = df_raw[df_raw['PRODUCCION'] == 0][['FECHA', 'PRODUCCION', 'VENTAS', 'DEVOLUCION' if 'DEVOLUCION' in df_raw.columns else 'DEVOLUCIONES']]
print(f"⚠️  Días con producción = 0 (excluidos del modelo):")
print(dias_cierre.to_string(index=False))

# Filtrar: eliminar días con cero producción
df = df_raw[df_raw['PRODUCCION'] > 0].copy()

# Crear variable DEMANDA REAL = VENTAS + DEVOLUCIONES
# Esto representa la demanda verdadera del mercado
dev_col = 'DEVOLUCION' if 'DEVOLUCION' in df.columns else 'DEVOLUCIONES'
df['DEMANDA'] = df['VENTAS'] + df[dev_col]

# Establecer FECHA como índice de la serie temporal
df = df.set_index('FECHA')
df.index.freq = None  # Serie diaria sin frecuencia forzada

print(f"\n✅ Serie limpia: {len(df)} días hábiles de operación")
print(f"   Desde: {df.index.min().date()}  →  Hasta: {df.index.max().date()}")
print(f"\n📊 Estadísticas de DEMANDA:")
print(df['DEMANDA'].describe().round(2))

# ─────────────────────────────────────────────────────────────────────────────
# CELDA 6 — DETECCIÓN DE OUTLIERS (±3 DESVIACIONES ESTÁNDAR)
# Según Hammam et al. (2025): "any data point beyond ±3 standard deviations
# from the rolling mean was excluded".
# Visualizamos los outliers ANTES de decidir qué hacer con ellos.
# ─────────────────────────────────────────────────────────────────────────────

# Calcular media y desviación estándar móviles (ventana 30 días)
rolling_mean = df['DEMANDA'].rolling(window=30, center=True).mean()
rolling_std  = df['DEMANDA'].rolling(window=30, center=True).std()

# Límites de ±3 desviaciones estándar
limite_sup = rolling_mean + 3 * rolling_std
limite_inf = rolling_mean - 3 * rolling_std

# Identificar outliers
outliers_mask = (df['DEMANDA'] > limite_sup) | (df['DEMANDA'] < limite_inf)
outliers = df[outliers_mask][['DEMANDA', 'VENTAS']]
outliers[dev_col] = df[outliers_mask][dev_col]

print(f"🔍 Outliers detectados: {outliers_mask.sum()} días")
if len(outliers) > 0:
    print(outliers.to_string())

# Visualización
fig, ax = plt.subplots(figsize=(16, 5))
ax.plot(df.index, df['DEMANDA'], color='#2196F3', linewidth=0.8, label='Demanda diaria')
ax.fill_between(df.index, limite_inf, limite_sup, alpha=0.15, color='gray', label='Banda ±3σ')
ax.scatter(df.index[outliers_mask], df['DEMANDA'][outliers_mask],
           color='red', s=40, zorder=5, label=f'Outliers ({outliers_mask.sum()})')
ax.set_title('Serie de Demanda con Banda de ±3 Desviaciones Estándar', fontsize=14, fontweight='bold')

# ─────────────────────────────────────────────────────────────────────────────
# CELDA 10 — PRUEBA DE ESTACIONARIEDAD (ADF TEST)
# La prueba ADF (Augmented Dickey-Fuller) determina si la serie es estacionaria.
# - Si p-valor < 0.05 → la serie ES estacionaria (no necesita diferenciación)
# - Si p-valor > 0.05 → la serie NO es estacionaria (ARIMA necesitará d>0)
# Esto es el paso 2 del framework de Hammam et al. (2025).
# ─────────────────────────────────────────────────────────────────────────────

serie = df['DEMANDA']

resultado_adf = adfuller(serie.dropna(), autolag='AIC')
print("=" * 55)
print("  TEST DE ESTACIONARIEDAD — ADF")
print("=" * 55)
print(f"  Estadístico ADF  : {resultado_adf[0]:.4f}")
print(f"  p-valor          : {resultado_adf[1]:.4f}")
print(f"  Lags usados      : {resultado_adf[2]}")
print(f"  Observaciones    : {resultado_adf[3]}")
print(f"\n  Valores críticos:")
for clave, valor in resultado_adf[4].items():
    print(f"    {clave}: {valor:.4f}")
print("=" * 55)
if resultado_adf[1] < 0.05:
    print("  ✅ CONCLUSIÓN: Serie ESTACIONARIA (p < 0.05)")
    print("     → Se puede usar ARIMA directamente")
    ES_ESTACIONARIA = True
else:
    print("  ⚠️  CONCLUSIÓN: Serie NO ESTACIONARIA (p ≥ 0.05)")
    print("     → ARIMA necesitará diferenciación (d ≥ 1)")
    ES_ESTACIONARIA = False

# ─────────────────────────────────────────────────────────────────────────────
# CELDA 11 — DESCOMPOSICIÓN ESTACIONAL Y GRÁFICOS ACF/PACF

# ─────────────────────────────────────────────────────────────────────────────

# Descomposición estacional (periodo = 7 días → semana)
descomposicion = seasonal_decompose(serie, model='additive', period=7, extrapolate_trend='freq')

axes[1].set_ylabel('Unidades')

descomposicion.seasonal.plot(ax=axes[2], color='#F57C00')
axes[2].set_title('Estacionalidad Semanal (periodo=7)', fontweight='bold')
axes[2].set_ylabel('Efecto estacional')

descomposicion.resid.plot(ax=axes[3], color='#C62828')
axes[3].set_title('Residuos (ruido no explicado)', fontweight='bold')
axes[3].set_ylabel('Residuo')

for ax in axes:
    ax.xaxis.set_major_formatter(mdates.DateFormatter('%Y'))
    ax.xaxis.set_major_locator(mdates.YearLocator())
    ax.grid(axis='y', linestyle='--', alpha=0.3)

plt.suptitle('Descomposición Estacional — Demanda Periódico Expreso',
             fontsize=14, fontweight='bold', y=1.01)
plt.tight_layout()
plt.savefig('fig_04_descomposicion.png', dpi=150, bbox_inches='tight')
plt.show()

# ACF y PACF
fig, axes = plt.subplots(1, 2, figsize=(16, 4))
plot_acf(serie.dropna(), lags=30, ax=axes[0], color='#1565C0')
axes[0].set_title('ACF — Autocorrelación (identifica q y estacionalidad)', fontweight='bold')
plot_pacf(serie.dropna(), lags=30, ax=axes[1], color='#388E3C', method='ywm')
axes[1].set_title('PACF — Autocorrelación Parcial (identifica p)', fontweight='bold')
plt.tight_layout()
plt.savefig('fig_05_acf_pacf.png', dpi=150, bbox_inches='tight')
plt.show()
print("📊 Gráficos guardados: fig_04_descomposicion.png, fig_05_acf_pacf.png")

# ─────────────────────────────────────────────────────────────────────────────
# CELDA 12 — SELECCIÓN AUTOMÁTICA DEL MODELO ARIMA/SARIMA
# Usamos auto_arima de pmdarima para encontrar los mejores parámetros
# (p, d, q) y estacionales (P, D, Q, s) minimizando el criterio AIC.
# Se prueban dos versiones: con y sin stepwise (como en Hammam et al.).
# ─────────────────────────────────────────────────────────────────────────────

print("🔍 Buscando mejor modelo SARIMA (puede tomar 1-3 minutos)...")
print("   Esto prueba múltiples combinaciones de parámetros automáticamente.\n")

# Modelo M1: CON función stepwise (más rápido)
modelo_m1 = pm.auto_arima(
    serie,
    start_p=0, max_p=3,
    start_q=0, max_q=3,
    d=None,           # auto-detectar diferenciación
    seasonal=True,
    m=7,              # periodo estacional = 7 días (semanal)
    start_P=0, max_P=2,
    start_Q=0, max_Q=2,
    D=None,
    stepwise=True,    # búsqueda inteligente (más rápida)
    information_criterion='aic',
    trace=False,
    error_action='ignore',
    suppress_warnings=True
)

# Modelo M2: SIN función stepwise (búsqueda más exhaustiva)
modelo_m2 = pm.auto_arima(
    serie,
    start_p=0, max_p=3,
    start_q=0, max_q=3,
    d=None,
    seasonal=True,
    m=7,
    start_P=0, max_P=2,
    start_Q=0, max_Q=2,
    D=None,
    stepwise=False,   # búsqueda completa
    information_criterion='aic',
    trace=False,
    error_action='ignore',
    suppress_warnings=True,
    n_jobs=-1
)

print("=" * 60)
print("  COMPARACIÓN DE MODELOS SARIMA (AIC y BIC)")
print("=" * 60)
print(f"  M1 (con stepwise)   → Orden: {modelo_m1.order} | Estacional: {modelo_m1.seasonal_order}")
print(f"     AIC: {modelo_m1.aic():.2f}  |  BIC: {modelo_m1.bic():.2f}")
print(f"\n  M2 (sin stepwise)   → Orden: {modelo_m2.order} | Estacional: {modelo_m2.seasonal_order}")
print(f"     AIC: {modelo_m2.aic():.2f}  |  BIC: {modelo_m2.bic():.2f}")

# Seleccionar el modelo con menor AIC
if modelo_m1.aic() <= modelo_m2.aic():
    mejor_sarima_info = modelo_m1
    print(f"\n  ✅ GANADOR: M1 (stepwise) — AIC más bajo")
else:
    mejor_sarima_info = modelo_m2
    print(f"\n  ✅ GANADOR: M2 (sin stepwise) — AIC más bajo")

ORDEN_SARIMA = mejor_sarima_info.order
ORDEN_ESTACIONAL = mejor_sarima_info.seasonal_order
print(f"     Parámetros seleccionados: SARIMA{ORDEN_SARIMA}x{ORDEN_ESTACIONAL}")
print("=" * 60)

# ─────────────────────────────────────────────────────────────────────────────

# CELDA 16 — INGENIERÍA DE FEATURES PARA XGBOOST
# XGBoost no entiende series de tiempo directamente.
# Debemos crear variables (features) que capturen:
#   1. Lags (valores pasados de la demanda)
#   2. Día de semana (0=Lunes, 6=Domingo)
#   3. Mes del año
#   4. Número de semana
#   5. Impacto de eventos del calendario
#   6. Tendencia (número de día desde el inicio)
# Esto sigue el paso de "Features extraction" de Hammam et al. (2025).
# ─────────────────────────────────────────────────────────────────────────────

def crear_features(df_input, lags=[1, 2, 3, 5, 7, 14], evento_serie=None):
    """
    Crea las variables de entrada para XGBoost.
    Parámetros:
        df_input: DataFrame con columna DEMANDA
        lags: lista de periodos de rezago a incluir
        evento_serie: Serie de pandas con impacto de eventos (índice=fecha)
    """
    df_feat = df_input[['DEMANDA']].copy()
    df_feat.columns = ['y']

    # Features temporales
    df_feat['dia_semana']  = df_feat.index.dayofweek     # 0=Lunes, 6=Domingo
    df_feat['es_finde']    = (df_feat['dia_semana'] >= 5).astype(int)  # 1 si sáb/dom
    df_feat['mes']         = df_feat.index.month
    df_feat['semana_anio'] = df_feat.index.isocalendar().week.astype(int)
    df_feat['dia_anio']    = df_feat.index.dayofyear
    df_feat['tendencia']   = np.arange(len(df_feat))     # progresión lineal del tiempo

    # Lags de demanda (valores de días anteriores)
    for lag in lags:
        df_feat[f'lag_{lag}'] = df_feat['y'].shift(lag)

    # Media móvil de 7 días (demanda promedio semanal anterior)
    df_feat['media_movil_7']  = df_feat['y'].shift(1).rolling(window=7).mean()
    df_feat['media_movil_14'] = df_feat['y'].shift(1).rolling(window=14).mean()

    # Impacto de eventos del calendario
    if evento_serie is not None:
        df_feat['impacto_evento'] = df_feat.index.map(
            lambda d: evento_serie.get(d, 0)
        )
    else:
        df_feat['impacto_evento'] = 0

    # Eliminar filas con NaN (generados por los lags)
    df_feat = df_feat.dropna()

    return df_feat

# Crear features para todo el dataset

# Buscamos los mejores hiperparámetros usando GridSearchCV.
# Los 6 hiperparámetros clave son los mismos de Hammam et al. (2025), Tabla 1.

    'max_depth'        : [3, 5],
    'n_estimators'     : [100, 200, 300],

# CELDA 19 — MODELO ENSEMBLE WAE (WEIGHTED AVERAGE ENSEMBLE)
# Combinamos SARIMA + XGBoost asignando pesos que minimizan el RMSE.
# Fórmula: F_WAE = ω1 × F_SARIMA + ω2 × F_XGBoost  |  ω1 + ω2 = 1
# Búsqueda por grid con pasos de 0.1 (igual que Hammam et al., 2025).
# ─────────────────────────────────────────────────────────────────────────────

# Alinear predicciones de SARIMA y XGBoost al mismo período de prueba
# (pueden tener longitudes distintas por diferencia en manejo de lags)
n_test_comun = min(len(pred_sarima), len(pred_xgb))
y_test_comun = test_sarima.values[-n_test_comun:]
pred_sarima_comun = pred_sarima[-n_test_comun:]
pred_xgb_comun    = pred_xgb[-n_test_comun:]

# Búsqueda de pesos óptimos
mejor_rmse_wae = np.inf
mejor_w1 = 0.5
mejor_w2 = 0.5

pesos = np.arange(0.0, 1.1, 0.1)
for w1 in pesos:
    w2 = round(1 - w1, 1)
    pred_wae_temp = w1 * pred_sarima_comun + w2 * pred_xgb_comun
    rmse_temp = np.sqrt(mean_squared_error(y_test_comun, pred_wae_temp))
    if rmse_temp < mejor_rmse_wae:
        mejor_rmse_wae = rmse_temp
        mejor_w1 = w1
        mejor_w2 = w2

# Predicción final del WAE
pred_wae = mejor_w1 * pred_sarima_comun + mejor_w2 * pred_xgb_comun
pred_wae = np.maximum(pred_wae, 0)

print("=" * 55)
print("  MODELO ENSEMBLE WAE — PESOS ÓPTIMOS")
print("=" * 55)
print(f"  ω1 (SARIMA)  : {mejor_w1:.1f}")
print(f"  ω2 (XGBoost) : {mejor_w2:.1f}")
print(f"  RMSE WAE     : {mejor_rmse_wae:.2f} unidades")

# Interpretación de los pesos
if mejor_w2 > mejor_w1:
    print(f"\n  📌 XGBoost tiene más peso → los datos tienen componente no lineal")
    print(f"     (similar a DS1 y DS4 en Hammam et al., 2025)")
elif mejor_w1 > mejor_w2:
    print(f"\n  📌 SARIMA tiene más peso → la demanda sigue patrón principalmente lineal")
    print(f"     (similar a DS3 en Hammam et al., 2025)")
else:
    print(f"\n  📌 Pesos iguales → mezcla equilibrada de linealidad y no linealidad")
    print(f"     (similar a DS2 en Hammam et al., 2025)")

# CELDA 20 — EVALUACIÓN Y COMPARACIÓN DE LOS TRES MODELOS
# Calculamos MAE, MSE, RMSE y MAPE para cada modelo.
# Los comparamos entre sí y contra los benchmarks de Hammam et al. (2025).
# ─────────────────────────────────────────────────────────────────────────────

def calcular_metricas(y_real, y_pred, nombre):
    """Calcula y retorna las 4 métricas estándar de pronóstico."""
    mae  = mean_absolute_error(y_real, y_pred)
    mse  = mean_squared_error(y_real, y_pred)
    rmse = np.sqrt(mse)
    # MAPE: evitar división por cero
    mask = y_real != 0
    mape = np.mean(np.abs((y_real[mask] - y_pred[mask]) / y_real[mask])) * 100
    return {'Modelo': nombre, 'MAE': round(mae, 2), 'MSE': round(mse, 2),
            'RMSE': round(rmse, 2), 'MAPE (%)': round(mape, 2)}

    calcular_metricas(y_test_comun, pred_xgb_comun,    'XGBoost'),
    calcular_metricas(y_test_comun, pred_wae,           'WAE (Ensemble)'),
])

MARGEN_SEGURIDAD   = 0.05  # 5% sobre el pronóstico. Cambia si quieres más/menos colchón

# Generamos fechas futuras (días hábiles, excluyendo domingos si aplica)
ultima_fecha = df.index.max()
fechas_futuras = pd.date_range(
    start=ultima_fecha + pd.Timedelta(days=1),
    periods=DIAS_A_PRONOSTICAR,
    freq='D'
)

# Pronóstico con SARIMA
pred_futura_sarima = resultado_sarima.forecast(steps=DIAS_A_PRONOSTICAR)
pred_futura_sarima = np.maximum(pred_futura_sarima, 0)

# Pronóstico con XGBoost: necesitamos construir features para el futuro
# Usamos el último valor conocido como punto de partida (rolling forecast)
demanda_hist = serie.values.copy()
pred_futura_xgb = []

for i, fecha in enumerate(fechas_futuras):
    # Construir un vector de features para este día futuro
    lag_vals = {}
    for lag in [1, 2, 3, 5, 7, 14]:
        idx = len(demanda_hist) - lag
        lag_vals[f'lag_{lag}'] = demanda_hist[idx] if idx >= 0 else demanda_hist.mean()

    feat_fila = {
        'dia_semana'   : fecha.dayofweek,
        'es_finde'     : int(fecha.dayofweek >= 5),
        'mes'          : fecha.month,
        'semana_anio'  : fecha.isocalendar()[1],
        'dia_anio'     : fecha.dayofyear,
        'tendencia'    : len(df) + i,
        **lag_vals,
        'media_movil_7' : np.mean(demanda_hist[-7:]),
        'media_movil_14': np.mean(demanda_hist[-14:]),
        'impacto_evento': eventos_serie.get(fecha.normalize(), 0),
    }

    X_fila = pd.DataFrame([feat_fila])[feature_cols]
    pred_i = max(0, float(modelo_xgb.predict(X_fila)[0]))
    pred_futura_xgb.append(pred_i)
    demanda_hist = np.append(demanda_hist, pred_i)

pred_futura_xgb = np.array(pred_futura_xgb)

# Ensemble WAE futuro
pred_futura_wae = mejor_w1 * pred_futura_sarima + mejor_w2 * pred_futura_xgb
pred_futura_wae = np.maximum(pred_futura_wae, 0)

# Seleccionar predicción del modelo ganador
if ganador == 'SARIMA':
    pred_final_futuro = pred_futura_sarima
elif ganador == 'XGBoost':
    pred_final_futuro = pred_futura_xgb
else:
    pred_final_futuro = pred_futura_wae

# Calcular producción recomendada
produccion_recomendada = np.ceil(pred_final_futuro * (1 + MARGEN_SEGURIDAD)).astype(int)

# Crear tabla de resultados
dias_semana_es = ['Lun', 'Mar', 'Mié', 'Jue', 'Vie', 'Sáb', 'Dom']
tabla_pronostico = pd.DataFrame({
    'Fecha'                 : fechas_futuras.strftime('%Y-%m-%d'),
    'Día'                   : [dias_semana_es[d.weekday()] for d in fechas_futuras],
    'Demanda Pronosticada'  : pred_final_futuro.round(0).astype(int),
    'Producir (recomendado)': produccion_recomendada,
    'Margen aplicado'       : [f"+{int(MARGEN_SEGURIDAD*100)}%" for _ in fechas_futuras],
    'Evento especial'       : [
        {-2: '🔴 Baja extrema', -1: '🟠 Baja', 0: '', 1: '🟢 Alta', 2: '🔵 Alta extrema'}.get(
            eventos_serie.get(d.normalize(), 0), ''
        )
        for d in fechas_futuras
    ]
})

print(f"\n{'='*80}")
print(f"  PLAN DE PRODUCCIÓN RECOMENDADO — Próximos {DIAS_A_PRONOSTICAR} días")
print(f"  Modelo: {ganador}  |  Margen de seguridad: {int(MARGEN_SEGURIDAD*100)}%")
print(f"{'='*80}")
print(tabla_pronostico.to_string(index=False))
print(f"{'='*80}")
print(f"\n  Total unidades a producir en los próximos {DIAS_A_PRONOSTICAR} días: "
      f"{produccion_recomendada.sum():,}")
print(f"  Promedio diario recomendado: {int(produccion_recomendada.mean()):,}")
